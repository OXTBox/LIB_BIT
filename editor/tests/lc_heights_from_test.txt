-- the Heights tab's From Floor / From Ceiling boxes (the owner's note, 18 Nov 2026).
-- Rebuilds BitEditor.livecode: run it ONLY in a copy of the editor folder, e.g.
--   python3 -I tests/py_fill_test.py   (writes tests/_fill/fill_test.cellmap)
--   copy editor/ to SCRATCH/LIB_BIT/editor, link SCRATCH/LIB_THEME to the real LIB_THEME, then
--   flock /tmp/oxt_headless.lock python3 -I OXT-BEYOND/tests/oxt_headless.py --root SCRATCH/LIB_BIT/editor --engine both tests/lc_heights_from_test.txt
local tD, tState, tText, tPath
repeat for each item tL in "lib_cellmap,build_editor,lib_bit,lib_theme,BitEditor"
  if there is a stack tL then
    if tL is not "BitEditor" then stop using stack tL
    delete stack tL
  end if
end repeat
put "/home/igame3d/projects/LIB_BIT/editor" into tD
start using stack (tD & "/livecode/build_editor.livecodescript")
Build.Editor tD
put ck("built", the result ends with "BitEditor.livecode", true) & return after tLog
send "Ed.Init" to stack "BitEditor"
set the defaultStack to "BitEditor"
send "Ed.OpenMap " & quote & tD & "/tests/_fill/fill_test.cellmap" & quote to stack "BitEditor"
send "Ed.MapTab Heights" to stack "BitEditor"
put ck("two checkboxes on the Heights tab", the style of button "HFromFloor" & "," & the style of button "HFromCeiling" & "," & the visible of group "HeightTools", "checkbox,checkbox,true") & return after tLog
put ck("their labels", the label of button "HFromFloor" & "/" & the label of button "HFromCeiling", "From Floor/From Ceiling") & return after tLog
put ck("an old map opens with both off", the hilite of button "HFromFloor" & "," & the hilite of button "HFromCeiling", "false,false") & return after tLog
put ck("they sit in the tool row, left of the hint", the right of button "HFromCeiling" < the left of field "HHint" and the top of button "HFromFloor" is 98, true) & return after tLog
put ck("coloured like the theme (template coloured before create)", the backgroundColor of button "HFromFloor" is not empty, true) & return after tLog
put value("Ed.State()", stack "BitEditor") into tState
send "mouseUp" to button "HFromFloor"
put ck("From Floor clicked: box on", the hilite of button "HFromFloor", true) & return after tLog
put ck("From Floor clicked: status says so", field "Status" contains "pillar", true) & return after tLog
send "mouseUp" to button "HFromCeiling"
put ck("From Ceiling clicked too: both on", the hilite of button "HFromFloor" & "," & the hilite of button "HFromCeiling", "true,true") & return after tLog
put ck("status: floor to top", field "Status" contains "floor to the top", true) & return after tLog
send "Ed.Undo" to stack "BitEditor"
put ck("one undo takes From Ceiling back", the hilite of button "HFromFloor" & "," & the hilite of button "HFromCeiling", "true,false") & return after tLog
send "Ed.Redo" to stack "BitEditor"
put ck("redo brings it back", the hilite of button "HFromCeiling", true) & return after tLog
put tD & "/tests/_fill/editor_saved.cellmap" into tPath
send "Ed.SaveMap " & quote & tPath & quote to stack "BitEditor"
put CellMap.ReadText(tPath) into tText
put ck("saved: fromCeiling 1 and fromFloor 1 in the map section", tText contains ("fromCeiling" & tab & "1" & return & "fromFloor" & tab & "1"), true) & return after tLog
send "mouseUp" to button "HFromFloor"
send "mouseUp" to button "HFromCeiling"
send "Ed.SaveMap " & quote & tPath & quote to stack "BitEditor"
put ck("both off again: no line in the file (as an old map)", CellMap.ReadText(tPath) contains "from", false) & return after tLog
send "Ed.OpenMap " & quote & tD & "/tests/_fill/fill_both.cellmap" & quote to stack "BitEditor"
send "Ed.MapTab Heights" to stack "BitEditor"
put ck("a saved map with both on opens with both boxes on", the hilite of button "HFromFloor" & "," & the hilite of button "HFromCeiling", "true,true") & return after tLog
return tLog
